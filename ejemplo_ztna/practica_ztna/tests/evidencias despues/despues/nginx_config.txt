    '{'
        '"fecha":"$time_iso8601",'
        '"id_peticion":"$request_id",'
        '"ip_origen":"$remote_addr",'
        '"metodo":"$request_method",'
        '"uri":"$request_uri",'
        '"estado":$status,'
        '"bytes":$body_bytes_sent,'
        '"tls_protocolo":"$ssl_protocol",'
        '"tls_cifrado":"$ssl_cipher",'
        '"cert_verificacion":"$ssl_client_verify",'
        '"cert_sujeto":"$ssl_client_s_dn",'
        '"cert_emisor":"$ssl_client_i_dn",'
        '"cert_serie":"$ssl_client_serial",'
        '"cert_caducidad":"$ssl_client_v_end",'
        '"backend":"$upstream_addr",'
        '"backend_estado":"$upstream_status",'
        '"tiempo":$request_time'
    '}';

# Marca como "denegada" toda respuesta 4xx o 5xx

    ~^[45]  1;

access_log ./nginx_denegados.log ztna_json if=$ztna_denegado;
